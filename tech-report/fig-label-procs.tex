      {(\{e_1',e_2'\}\uplus E,L)} }

    {\trans{(\{e_1,e_2\}\uplus E,L)}\tau
      {(\{e_1',e_2'\}\uplus E,L)} }

    {\{x,y\}\in L \\
      \trans{e_1}{\closel x}{e_1'} \and
      \trans{e_2}{\closel y}{e_2'} }

      {(\{e_1',e_2'\}\uplus E,L\setminus\{\{x,y\}\})} }
