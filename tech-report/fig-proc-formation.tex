  \declrel{Liveness/context agreement}
    {$\operator{LiveCtx}(L;\Gamma)$}
  \begin{equation*}
    \operator{LiveCtx}(L;\Gamma)
    \quad\Longleftrightarrow\quad
    \operator{ends}(L)=\operator{available}(\Gamma),
    \qquad
    \operator{ends}(L)=\{x\mid\exists y.\ \{x,y\}\in L\}.
  \end{equation*}

  \declrel{Splitting a configuration context}
    {$\Gamma\leadsto\Gamma_1\mid\Gamma_2$}
  \begin{mathpar}
    \axiom{S-Empty}
    {\Empty\leadsto\Empty\mid\Empty}

    \infrule{S-LinL}
    {\Gamma\leadsto\Gamma_1\mid\Gamma_2}
    {\Gamma,\ebind xS\leadsto
       \Gamma_1,\ebind xS\mid\Gamma_2,\ebind{x}{\bullet S}}

    \infrule{S-LinR}
    {\Gamma\leadsto\Gamma_1\mid\Gamma_2}
    {\Gamma,\ebind xS\leadsto
       \Gamma_1,\ebind{x}{\bullet S}\mid\Gamma_2,\ebind xS}

    \infrule{S-Used}
    {\Gamma\leadsto\Gamma_1\mid\Gamma_2}
    {\Gamma,\ebind{x}{\bullet S}\leadsto
       \Gamma_1,\ebind{x}{\bullet S}\mid\Gamma_2,\ebind{x}{\bullet S}}
  \end{mathpar}

  \declrel{Typing a thread multiset}{$\Gamma\vdash_{\mathsf{th}}E$}
  \begin{mathpar}
    \infrule{TM-Empty}
    {\operator{AllUsed}(\Gamma)}
    {\Gamma\vdash_{\mathsf{th}}\emptyset}

    \infrule{TM-Thread}
    {\Gamma\leadsto\Gamma_e\mid\Gamma_r \\
     \expAgainst[\Empty]{\Gamma_e}{e}{\TUnit}{\Gamma_e'} \\
     \operator{AllUsed}(\Gamma_e') \and
     \Gamma_r\vdash_{\mathsf{th}}E}
    {\Gamma\vdash_{\mathsf{th}}\{e\}\uplus E}
  \end{mathpar}

  \declrel{Typing a process configuration}{$\isProc K$}
  \begin{mathpar}
    \infrule{T-Conf}
    {\operator{LiveCtx}(L;\Gamma) \and
     \Gamma\vdash_{\mathsf{th}}E}
    {\isProc[\Gamma]{(E,L)}}
  \end{mathpar}
  \caption{Typing rules for process configurations.  The function
    $\operator{available}$ returns the variables with unmarked linear bindings;
    a bullet marks a binding that retains its type but is no longer available.}
  \label{fig:proc-formation}
\end{figure}
